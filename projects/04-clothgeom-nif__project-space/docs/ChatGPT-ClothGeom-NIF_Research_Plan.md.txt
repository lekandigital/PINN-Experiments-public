## Step 2: Latent Code Extraction Script

We now implement a Python script to extract **latent cloth states** (node positions and edge strains) from a trained HGNN and pair them with ground-truth SDF volumes. Dependencies include PyTorch, NumPy, and h5py for HDF5 export. The script loads HGNN outputs, computes strains, aligns with the SDF grid, and writes paired data:

```python
import torch
import numpy as np
import h5py

# Device configuration
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

def load_hgnn_output(file_path):
    """
    Load HGNN output containing cloth mesh states.
    Expects a PyTorch file with keys 'positions' (N_nodes x 3) and 'edges' (N_edges x 2).
    """
    data = torch.load(file_path, map_location=device)
    positions = data['positions'].to(device)  # Node positions (x,y,z)
    edges = data['edges'].to(device)          # Edge index pairs (u,v)
    return positions, edges

def compute_edge_strains(positions, edges, rest_lengths):
    """
    Compute per-edge strain = (current_length - rest_length) / rest_length.
    Args:
        positions (Tensor[N_nodes,3]): current node positions.
        edges (Tensor[N_edges,2]): indices of nodes for each edge.
        rest_lengths (Tensor[N_edges]): original (rest) lengths of edges.
    Returns:
        strains (Tensor[N_edges]): strain values for each edge.
    """
    # Coordinates of edge endpoints
    p1 = positions[edges[:, 0]]  # shape (N_edges,3)
    p2 = positions[edges[:, 1]]
    # Current lengths
    curr_lengths = torch.norm(p1 - p2, dim=1)
    # Strain calculation
    strains = (curr_lengths - rest_lengths) / rest_lengths
    return strains

# Load the rest (reference) edge lengths from simulation data
rest_lengths = torch.load('edge_rest_lengths.pt', map_location=device)  # shape [N_edges]

# Load HGNN output for one cloth sample
positions, edges = load_hgnn_output('hgnn_output.pt')

# Compute latent features: node positions and edge strains
strains = compute_edge_strains(positions, edges, rest_lengths)
latent_node_features = positions.view(-1)        # flatten all node (x,y,z) into 1D
latent_edge_features = strains.view(-1)          # flatten all edge strains
latent_code = torch.cat([latent_node_features, latent_edge_features], dim=0)  # (1D tensor)

# Load the corresponding ground-truth SDF volume (from simulator)
# For example, a NumPy file of shape (Nx,Ny,Nz) with signed distance values.
sdf_volume = np.load('ground_truth_sdf.npy')  # shape [Nx, Ny, Nz]

# Export paired data to HDF5 for training the decoder
with h5py.File('cloth_data_paired.h5', 'w') as h5f:
    h5f.create_dataset('latent_code', data=latent_code.cpu().numpy())
    h5f.create_dataset('sdf_volume', data=sdf_volume)
    # Optionally store metadata
    h5f.attrs['num_nodes'] = positions.shape[0]
    h5f.attrs['num_edges'] = edges.shape[0]
```

This script assumes the HGNN output (`hgnn_output.pt`) provides the current node positions and connectivity. It computes strains based on original edge lengths, flattens everything into a 1D latent vector, then pairs it with the SDF grid before saving to an HDF5 dataset.

## Step 3: NIF Decoder (PyTorch Implementation)

We implement the Neural Implicit Field (NIF) decoder as a sine-activated MLP with latent modulation. The network takes a 3D point coordinate and a latent code (extracted from Step 2) and outputs a signed distance **SDF value** and a **variance (density)** for level-of-detail blending. Key features:

- **Sine activations (SIREN-like):** We use periodic activations to capture high-frequency detail.
- **Latent conditioning:** The latent code is concatenated to the input and influences all layers.
- **Variance field:** The network also outputs a positive variance for LOD blending (e.g. for texture/mip-mapping).
- **Optimizations:** We note that in practice a multiresolution hash encoding (as in Instant-NGP) can be used for speed, and we set the model to FP16 precision for faster inference on GPU.

```python
import torch
import torch.nn as nn
import numpy as np

# Define a sine activation layer (for SIREN)
class Sine(nn.Module):
    def __init__(self, w0=1.0):
        super().__init__()
        self.w0 = w0
    def forward(self, x):
        return torch.sin(self.w0 * x)

class NIFDecoder(nn.Module):
    """
    Neural Implicit Field (NIF) decoder: sine-activated MLP with latent modulation.
    Input: 3D coords (x,y,z) and latent code of dimension latent_dim.
    Output: signed distance (sdf) and variance (density) for each point.
    """
    def __init__(self, latent_dim, num_layers=5, hidden_dim=256, w0=30.0):
        super().__init__()
        self.latent_dim = latent_dim
        self.input_dim = 3 + latent_dim  # 3 coords + latent features
        self.hidden_dim = hidden_dim

        # Build MLP layers with sine activations
        layers = []
        for i in range(num_layers):
            in_features = self.input_dim if i == 0 else hidden_dim
            lin = nn.Linear(in_features, hidden_dim)
            # SIREN initialization: uniform in [-sqrt(6/in)/w0, +sqrt(6/in)/w0]
            bound = np.sqrt(6 / in_features) / w0
            nn.init.uniform_(lin.weight, -bound, bound)
            layers.append(lin)
            layers.append(Sine(w0))
        self.net = nn.Sequential(*layers)

        # Final layers: one for SDF output, one for variance (density)
        self.sdf_out = nn.Linear(hidden_dim, 1)
        self.var_out = nn.Linear(hidden_dim, 1)

    def forward(self, coords, latent):
        """
        coords: Tensor[N,3], 3D points.
        latent: Tensor[latent_dim] or Tensor[N, latent_dim] latent code(s).
        Returns:
            sdf (Tensor[N,1]), variance (Tensor[N,1]).
        """
        # If latent is 1D, expand to match coords
        if latent.ndim == 1:
            latent = latent.unsqueeze(0).expand(coords.shape[0], -1)  # shape (N, latent_dim)
        x = torch.cat([coords, latent], dim=1)  # (N, 3+latent_dim)
        features = self.net(x)
        sdf = self.sdf_out(features)                 # Signed distance output
        variance = torch.exp(self.var_out(features)) # Positive variance (density)
        return sdf, variance
```

## Step 4: Cloth Mesh Generation Script

From the trained NIF decoder, we can obtain a full SDF field and extract an explicit mesh using Marching Cubes. The following script samples the SDF on a regular grid within given bounds and applies marching cubes at the zero level-set $SDF(\mathbf{x})=0$. We also demonstrate **progressive resolution** by extracting meshes at multiple grid resolutions. Dependencies: NumPy and scikit-image.

```python
import numpy as np
from skimage import measure

def extract_mesh_from_sdf(sdf_volume, bounds):
    """
    Extract a triangular mesh from a 3D SDF volume using marching cubes.
    sdf_volume: 3D numpy array of shape (Nx,Ny,Nz) of signed distances.
    bounds: ((xmin,xmax),(ymin,ymax),(zmin,zmax)) bounding box in world coords.
    Returns: verts (Vx3 array), faces (Fx3 array), normals (Vx3).
    """
    # Determine grid dimensions and spacing
    Nx, Ny, Nz = sdf_volume.shape
    (xmin,xmax), (ymin,ymax), (zmin,zmax) = bounds
    spacing = ((xmax-xmin)/(Nx-1), (ymax-ymin)/(Ny-1), (zmax-zmin)/(Nz-1))
    
    # Apply marching cubes at isosurface level=0.0
    verts, faces, normals, values = measure.marching_cubes(
        sdf_volume, level=0.0, spacing=spacing)
    # The verts are returned in world coordinates given the spacing.
    return verts, faces, normals
```

```python
import torch
```

```python
def compute_metrics(pred_mesh, gt_mesh):
    """
    Compute Chamfer distance and normal consistency between two PyTorch3D Meshes.

    # Sample points from meshes for Chamfer evaluation

cd_vals, nc_vals = [], []
for f in lod_factors:
    # For simplicity, assume we can refine mesh by subdividing
    # (In practice, one could decimate instead.)
    subdiv = int(1/f)
    pred_lod = pred_mesh.subdivide(subdiv)
    gt_lod   = gt_mesh.subdivide(subdiv)
    cd, nc = compute_metrics(pred_lod, gt_lod)
    cd_vals.append(cd)
    nc_vals.append(nc)

# Plot the metrics vs. LOD
plt.figure(figsize=(6,3))
plt.plot(lod_factors, cd_vals, 'o-', label='Chamfer')
plt.plot(lod_factors, nc_vals, 's-', label='Normal Consistency')
plt.xlabel("LOD Factor (smaller=coarser)")
plt.title("Reconstruction Metrics vs LOD")
plt.legend()
plt.show()
```

This script uses `chamfer_distance` and `mesh_normal_consistency` from PyTorch3D. It assumes access to `Meshes` objects for the predicted and ground-truth geometry. The results are plotted for visual analysis across resolutions.
