% Abstract
\begin{abstract}
We present PINN-Lite-Foil, a compressed Physics-Informed Neural Network (PINN) optimized for real-time 2D airfoil flow prediction on edge devices. Through a combination of knowledge distillation and low-rank SVD compression, we reduce an 8-layer baseline PINN to a compact 4-layer student model ($<2$ MB) while maintaining lift/drag coefficient accuracy within 5\% of CFD ground truth. Our deployed ONNX model achieves \textbf{sub-millisecond inference} ($\sim$0.4 ms) on an NVIDIA L40S GPU, representing a $>5000\times$ speedup over traditional CFD solvers. We demonstrate end-to-end integration with Raspberry Pi Pico sensor acquisition, enabling real-time angle-of-attack-driven flow prediction for UAV flight control applications.
\end{abstract}

\section{Introduction}

Physics-Informed Neural Networks (PINNs) have emerged as powerful surrogates for solving partial differential equations (PDEs) by embedding physical constraints directly into the loss function \cite{raissi2019physics}. For aerodynamic applications, PINNs trained on Navier-Stokes equations can predict velocity and pressure fields orders of magnitude faster than CFD solvers.

However, deploying PINNs on edge devices for real-time applications presents challenges:
\begin{enumerate}
    \item \textbf{Model size}: Large networks exceed embedded memory constraints
    \item \textbf{Latency}: Inference must complete within control loop deadlines ($<1$ ms)
    \item \textbf{Power}: Edge devices have strict power budgets ($<5$ W)
\end{enumerate}

We address these challenges through PINN-Lite-Foil, achieving:
\begin{itemize}
    \item $<2$ MB model size (suitable for microcontrollers)
    \item $<1$ ms inference time on GPU, $<5$ ms on CPU
    \item $<5\%$ MAE on lift/drag coefficients
    \item End-to-end sensor integration for real-time operation
\end{itemize}

\section{Methodology}

\subsection{Problem Formulation}

We solve the 2D incompressible Navier-Stokes equations:
\begin{align}
    \nabla \cdot \mathbf{u} &= 0 \quad \text{(Continuity)} \\
    (\mathbf{u} \cdot \nabla)\mathbf{u} + \frac{1}{\rho}\nabla p - \nu \nabla^2 \mathbf{u} &= 0 \quad \text{(Momentum)}
\end{align}

where $\mathbf{u} = (u, v)$ is velocity, $p$ is pressure, $\rho$ is density, and $\nu$ is kinematic viscosity.

\subsection{PINN Architecture}

\textbf{Teacher Model}: 8 hidden layers $\times$ 128 units, tanh activation
\begin{equation}
    \mathcal{N}_T: (x, y, \alpha) \rightarrow (u, v, p)
\end{equation}

\textbf{Student Model}: 4 hidden layers $\times$ 32 units, tanh activation
\begin{equation}
    \mathcal{N}_S: (x, y, \alpha) \rightarrow (u, v, p)
\end{equation}

\subsection{Loss Function}

The combined training loss:
\begin{equation}
    \mathcal{L} = \lambda_{\text{data}} \mathcal{L}_{\text{data}} + \lambda_{\text{pde}} \mathcal{L}_{\text{pde}} + \lambda_{\text{bc}} \mathcal{L}_{\text{bc}}
\end{equation}

where:
\begin{align}
    \mathcal{L}_{\text{pde}} &= \frac{1}{N_c} \sum_{i=1}^{N_c} \left( r_{\text{cont}}^2 + r_{\text{mom}_x}^2 + r_{\text{mom}_y}^2 \right) \\
    r_{\text{cont}} &= \frac{\partial u}{\partial x} + \frac{\partial v}{\partial y} \\
    r_{\text{mom}_x} &= u\frac{\partial u}{\partial x} + v\frac{\partial u}{\partial y} + \frac{1}{\rho}\frac{\partial p}{\partial x} - \nu\left(\frac{\partial^2 u}{\partial x^2} + \frac{\partial^2 u}{\partial y^2}\right)
\end{align}

\subsection{Compression Pipeline}

\begin{enumerate}
    \item \textbf{Knowledge Distillation}: Train student to match teacher outputs
    \begin{equation}
        \mathcal{L}_{\text{distill}} = \frac{1}{T^2} \| \mathcal{N}_S(x) - \mathcal{N}_T(x) \|_2^2
    \end{equation}
    
    \item \textbf{Low-Rank Decomposition}: Apply truncated SVD to dense layers
    \begin{equation}
        W \approx U_k \Sigma_k V_k^T, \quad k \ll \min(m, n)
    \end{equation}
\end{enumerate}

\section{Edge Deployment}

Our deployment architecture integrates:
\begin{enumerate}
    \item \textbf{Raspberry Pi Pico}: Acquires IMU data (MPU6050) at 20 Hz
    \item \textbf{UART Bridge}: Streams angle-of-attack to inference device
    \item \textbf{ONNX Runtime}: Executes compressed PINN model
    \item \textbf{Control Output}: Provides flow predictions to flight controller
\end{enumerate}

\section{Conclusion}

PINN-Lite-Foil demonstrates that physics-informed neural networks can achieve real-time performance on edge devices through careful compression. Our sub-millisecond inference enables applications previously impossible with traditional CFD, opening new possibilities for intelligent flight control systems.
