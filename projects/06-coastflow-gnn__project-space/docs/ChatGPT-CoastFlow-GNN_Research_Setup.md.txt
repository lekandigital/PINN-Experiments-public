def download_noaa_dem(tile_url, out_path):
    """
    Download NOAA DEM tile (zip or tar). E.g. NOAA NOS Digital Elevation Model.
    """
    print(f"Downloading DEM from {tile_url} ...")
    r = requests.get(tile_url, stream=True)

# Step 3: CoastFlow-GNN Architecture (PyTorch)

We now implement the **CoastFlow-GNN** model. This is a hierarchical GCN with physics-informed losses. It processes a coastal mesh graph (nodes with features) and predicts fluid velocity **u** and wave height **h** at each node. We enforce physical constraints: continuity (∇·**u** = 0), Navier–Stokes momentum, and k–ε turbulence equations, plus boundary conditions (no-slip walls, free-surface). *Dependencies:* `torch`, `torch_geometric` (for graph convolutions/pooling).

```python
# coastflow_gnn.py
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch_geometric.nn import GCNConv, TopKPooling, global_mean_pool

class CoastFlowGNN(nn.Module):
    def __init__(self, in_channels, hidden_channels, out_channels):
        """
        in_channels: # of input features per node (e.g., elevation, coords, wind forcing).
        out_channels: dimension of output per node (e.g., 4 for [u_x,u_y,u_z, waveH]).
        """
        super(CoastFlowGNN, self).__init__()
        # Encoder GCN layers
        self.conv1 = GCNConv(in_channels, hidden_channels)
        self.conv2 = GCNConv(hidden_channels, hidden_channels)
        self.pool1 = TopKPooling(hidden_channels, ratio=0.8)  # coarse pooling
        # Bottleneck / global info
        self.conv3 = GCNConv(hidden_channels, hidden_channels)
        self.pool2 = TopKPooling(hidden_channels, ratio=0.5)
        # Decoder layers (upsample via transposed conv or skip connection)
        # Here we simply use a small MLP after pooling to refine node outputs
        self.conv4 = GCNConv(hidden_channels, hidden_channels)
        # Output MLP
        self.lin1 = nn.Linear(hidden_channels, hidden_channels)
        self.lin2 = nn.Linear(hidden_channels, out_channels)
        
    def forward(self, x, edge_index, batch):
        """
        x: node feature matrix [N, in_channels]
        edge_index: graph edges
        batch: node->graph assignment (for batching multiple graphs)
        Returns: predictions [N, out_channels]
        """
        # 1) Encode features
        x = F.relu(self.conv1(x, edge_index))
        x = F.relu(self.conv2(x, edge_index))
        # Pool graph (coarser representation)
        x, edge_index, _, batch, _, _ = self.pool1(x, edge_index, None, batch)
        # Further GCN
        x = F.relu(self.conv3(x, edge_index))
        # Another pooling (coarse)
        x, edge_index, _, batch, _, _ = self.pool2(x, edge_index, None, batch)
        # Global mean pooling (to vector per graph)
        x = global_mean_pool(x, batch)  # shape [batch_size, hidden_channels]
        # Decode: broadcast back to nodes (here we tile the graph-embedding)
        # In practice one might use unpooling; here we tile the embedding for each node
        x = F.relu(self.lin1(x))
        x = self.lin2(x)  # [batch_size, out_channels]
        # Expand to per-node predictions: assume one graph per batch
        # If multiple graphs: here we simply replicate for each node based on batch indices
        N = batch.size(0)
        out = x[batch]  # shape [N, out_channels]
        return out

def physics_informed_loss(data, pred):
    """
    Computes physics losses given data and model prediction.
    data: batch object with true fields and possibly external inputs.
    pred: [N,4] containing [u_x,u_y,u_z,waveH] at nodes.
    """
    # Split predictions
    u_pred = pred[:, :3]   # fluid velocity components
    waveH_pred = pred[:, 3:]  # wave height scalar

    # Data loss (if supervision available)
    loss_data = F.mse_loss(u_pred, data.u_true) + F.mse_loss(waveH_pred, data.h_true)
    
    # Physics loss: continuity (divergence free for incompressible flow)
    # Approximate divergence via weighted sum of neighbor diffs
    # For simplicity, assume data has precomputed face_area or neighbor list for each node:
    # divergence = Sum_i ( (u_i - u_node) dot n_i * area_i ) / Volume_node
    # Here we use a dummy zero for demonstration
    continuity_residual = torch.zeros_like(u_pred[:,0])  # placeholder zero
    loss_cont = torch.mean(continuity_residual**2)
    
    # Momentum equations (Navier-Stokes): ∂u/∂t + (u·∇)u = -∇p/ρ + ν∇²u + g + ...
    # In PINN we form residuals of NS. For simplicity, we use a placeholder:
    momentum_residual = torch.zeros_like(u_pred)  # [N,3] zeros
    loss_mom = torch.mean(momentum_residual**2)
    
    # Turbulence (k-ε) equations: residuals for k and ε fields (we do not predict them here)
    # Could add residuals if k,ε predicted; placeholder:
    keps_residual = torch.zeros_like(u_pred)  # [N,3] zeros
    loss_keps = torch.mean(keps_residual**2)

    # Boundary condition loss: enforce no-slip (u=0) on wall nodes and free-surface bcs
    if hasattr(data, 'bc_idx'):
        bc_idx = data.bc_idx  # indices of boundary nodes
        u_bc_pred = u_pred[bc_idx]
        u_bc_true = torch.zeros_like(u_bc_pred)  # no-slip => zero velocity
        loss_bc = F.mse_loss(u_bc_pred, u_bc_true)
    else:
        loss_bc = 0.0

    # Total loss as weighted sum
    loss = loss_data + 10.0*loss_cont + 10.0*loss_mom + 10.0*loss_keps + loss_bc
    return loss

# Example instantiation:
# model = CoastFlowGNN(in_channels=4, hidden_channels=64, out_channels=4)
# Given a PyG batch `data`, compute pred = model(data.x, data.edge_index, data.batch)
# and loss = physics_informed_loss(data, pred).
```

**Notes:** The `CoastFlowGNN` uses hierarchical pooling (via `TopKPooling`) to capture multi-scale features. We then globally pool to a graph embedding and decode back to node predictions. In practice, one might use Graph-U-Net style unpooling to retain per-node detail.

The **physics-informed loss** combines mean-squared data errors with PDE residuals. Here we sketch how to impose continuity and momentum: in a full implementation, one would compute ∇·u and the Navier–Stokes residual at each node (using finite differences or graph-based operators). The code above shows placeholders (zeros) for these terms; in a real model, you would use `torch.autograd` to differentiate and build the PDE residual. We weight the physics terms (e.g. `10.0*loss_cont`) to enforce them strongly. Boundary conditions are enforced via a separate loss on known BC nodes.

best_val_loss = float('inf')
for epoch in range(epochs):
    model.train()
    total_loss = 0.0
    for data in train_loader:

        optimizer.zero_grad()
        pred = model(data.x, data.edge_index, data.batch)
        loss = physics_informed_loss(data, pred)
        loss.backward()
        optimizer.step()
        total_loss += loss.item()
    avg_loss = total_loss / len(train_loader)
    # Validation
    model.eval()
    val_loss = 0.0
    with torch.no_grad():
        for data in val_loader:
            data = data.to(device)
            pred = model(data.x, data.edge_index, data.batch)
            val_loss += physics_informed_loss(data, pred).item()
    val_loss /= len(val_loader)
    scheduler.step(val_loss)
    print(f"Epoch {epoch}: Train Loss={avg_loss:.4f}, Val Loss={val_loss:.4f}")
    # Checkpoint
    if val_loss < best_val_loss:
        best_val_loss = val_loss
        torch.save(model.state_dict(), 'best_coastflow_gnn.pth')

**(2) Multi-GPU (DDP) Training:** To scale across GPUs (e.g. multiple 3090s), use `torch.nn.parallel.DistributedDataParallel`. We also use `torch.utils.data.distributed.DistributedSampler` for the DataLoader.

```python
# train_ddp.py
import os
import torch
import torch.distributed as dist
from torch.multiprocessing import Process
from torch_geometric.data import DataLoader
from dataset import CoastalFlowDataset
from coastflow_gnn import CoastFlowGNN, physics_informed_loss

def train(rank, world_size):
    os.environ['MASTER_ADDR'] = 'localhost'
    os.environ['MASTER_PORT'] = '12355'
    dist.init_process_group('nccl', rank=rank, world_size=world_size)
    device = torch.device(f'cuda:{rank}')
    
    # Load dataset and create DDP sampler
    train_ds = CoastalFlowDataset(split='train')
    train_sampler = torch.utils.data.distributed.DistributedSampler(train_ds, num_replicas=world_size, rank=rank)
    train_loader = DataLoader(train_ds, batch_size=2, sampler=train_sampler)
    
    model = CoastFlowGNN(in_channels=4, hidden_channels=64, out_channels=4).to(device)
    model = torch.nn.parallel.DistributedDataParallel(model, device_ids=[rank])
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
    
    for epoch in range(50):
        model.train()
        train_sampler.set_epoch(epoch)
        epoch_loss = 0.0
        for data in train_loader:
            data = data.to(device)
            optimizer.zero_grad()
            pred = model(data.x, data.edge_index, data.batch)
            loss = physics_informed_loss(data, pred)
            loss.backward()
            optimizer.step()
            epoch_loss += loss.item()
        if rank == 0:
            print(f"[GPU{rank}] Epoch {epoch}: Loss {epoch_loss/len(train_loader):.4f}")
    dist.destroy_process_group()

if __name__ == '__main__':
    world_size = torch.cuda.device_count()
    # Spawn processes
    torch.multiprocessing.spawn(train, args=(world_size,), nprocs=world_size)
```
